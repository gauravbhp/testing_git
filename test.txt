def fetch_kit_elements(code, pallet_number, box_number=None):
    """
    Fetch kit elements from SKP_UPLOAD table
    """
    print(f"[KIT QUERY] Fetching kit elements")
    print(f"[KIT QUERY] Code: {code}")
    print(f"[KIT QUERY] Palette: {pallet_number}")
    
    try:
        # Agar box_number nahi diya toh pehle fetch karo
        if box_number is None:
            box_sql = """
            SELECT DISTINCT BOXNUMBER
            FROM SKP_UPLOAD
            WHERE CODE = ?
            AND PALLETNUMBER = ?
            ORDER BY BOXNUMBER
            """
            
            cursor.execute(box_sql, (code, pallet_number))
            box_results = cursor.fetchall()
            
            if not box_results:
                print(f"[KIT QUERY] No box numbers found for code: {code}")
                return []
            
            box_numbers = [row[0] for row in box_results]
            print(f"[KIT QUERY] Found box numbers: {box_numbers}")
            box_number = box_numbers[0]
            print(f"[KIT QUERY] Using first box number: {box_number}")
        
        # Fetch elements for this box
        sql = """
        SELECT
            COMPANYCODE, PARTDESC, TOTALPCS, ITEMTYPECODE,
            DECOSUBCODE01, DECOSUBCODE02, DECOSUBCODE03, DECOSUBCODE04,
            DECOSUBCODE05, DECOSUBCODE06, DECOSUBCODE07, DECOSUBCODE08,
            DECOSUBCODE09, DECOSUBCODE10, CODE, ELEMENTDESC, BOXNUMBER,
            ELEMENTSEQ, PALLETNUMBER, WEIGHTUNITOFMEASURECODE, 
            NETWEIGHT, GROSSWEIGHT, CREATIONDATETIME, CREATIONUSER,
            ABSUNIQUEID, BOXSEQUENCE, PACKINGSEQUENCE, PLACEMENTINBOX
        FROM SKP_UPLOAD
        WHERE CODE = ?
        AND PALLETNUMBER = ?
        AND BOXNUMBER = ?
        ORDER BY ELEMENTSEQ
        """
        
        cursor.execute(sql, (code, pallet_number, box_number))
        results = cursor.fetchall()
        
        # Convert to list of dicts
        kit_elements = []
        for row in results:
            kit_elements.append({
                'companycode': row[0],
                'partdesc': row[1],
                'totalpcs': row[2],
                'itemtypecode': row[3],
                'decosubcode01': row[4],
                'decosubcode02': row[5],
                'decosubcode03': row[6],
                'decosubcode04': row[7],
                'decosubcode05': row[8],
                'decosubcode06': row[9],
                'decosubcode07': row[10],
                'decosubcode08': row[11],
                'decosubcode09': row[12],
                'decosubcode10': row[13],
                'code': row[14],
                'elementdesc': row[15],
                'boxnumber': row[16],
                'elementseq': row[17],
                'palletnumber': row[18],
                'weightunit': row[19],
                'netweight': row[20],
                'grossweight': row[21],
                'creationdatetime': row[22],
                'creationuser': row[23],
                'absuniqueid': row[24],
                'boxsequence': row[25],
                'packingsequence': row[26],
                'placementinbox': row[27]
            })
        
        print(f"[KIT QUERY] Found {len(kit_elements)} kit elements")
        return kit_elements
        
    except Exception as e:
        print(f"[KIT ERROR] {str(e)}")
        return []
