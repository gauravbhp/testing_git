// Fetch element data when select changes
function fetchElementData(elementId) {
    if (!elementId) {
        document.getElementById('elementInputs').innerHTML = '';
        return;
    }

    // Show loading state
    document.getElementById('elementInputs').innerHTML = '<div class="text-center py-4">Loading...</div>';

    // Fetch data from server
    fetch(
`/get_element_data/${encodeURIComponent(elementId)}/?order_code=${orderCode}&demand_code=${demandCode}&pallet_number=${palletNumber}&box_number=${boxNumber}`,{
        headers: {
            'X-Requested-With': 'XMLHttpRequest'
        }
    })
        .then(response => {
            const contentType = response.headers.get('content-type') || '';
            if (contentType.includes('application/json')) {
                return response.json();
            }
            return response.text().then(text => {
                throw new Error(text.replace(/<[^>]+>/g, '').slice(0, 300));
            });
        })
        .then(data => {
            if (data.status === 'success') {
                elementData = data.element_data || {};
                baselineData = data.baseline_data || {};
                toleranceData = data.tolerance_data || {};
                renderInputFields();
            } else {
                console.error('Element data fetch failed:', data);
                document.getElementById('elementInputs').innerHTML = '<div class="text-red-500 py-4">Error: ' + (data.message || 'Unknown error') + '</div>';
            }
        })
        .catch(error => {
            console.error('Error fetching element data:', error);
            document.getElementById('elementInputs').innerHTML = '<div class="text-red-500 py-4">Error loading data: ' + error.message + '</div>';
        });
}

function formatToleranceText(fieldId, tolerance) {
    const baselineValue = parseFloat(baselineData[fieldId]);
    const parsedTolerance = parseFloat(tolerance);

    if (!isNaN(parsedTolerance)) {
        if (!isNaN(baselineValue)) {
            const min = baselineValue - parsedTolerance;
            const max = baselineValue + parsedTolerance;
            return `Baseline ${baselineValue}, tolerance ${parsedTolerance}: allowed range ${min} - ${max}`;
        }
        return `Tolerance: ±${parsedTolerance}`;
    }
    return 'Tolerance: N/A';
}

// Validate field against tolerance
function validateField(fieldId) {
    const input = document.getElementById(`input_${fieldId}`);
    const errorDiv = document.getElementById(`error_${fieldId}`);
    const tolInfo = document.getElementById(`tol_info_${fieldId}`);
    const submitBtn = document.getElementById('submitBtn');
    
    if (!input || !errorDiv) return true;

    const value = parseFloat(input.value);
    const field = FIELD_CONFIG.find(f => f.id === fieldId);
    const tolerance = parseFloat(toleranceData[field.tolField]);
    const baseline = parseFloat(baselineData[field.id]);

    // Reset error state
    errorDiv.classList.add('hidden');
    input.classList.remove('border-red-500', 'border-green-500');

    if (tolInfo) {
        tolInfo.textContent = formatToleranceText(fieldId, tolerance);
    }

    if (isNaN(value)) {
        if (submitBtn) {
            const allValid = checkAllFieldsValid();
            submitBtn.disabled = !allValid;
        }
        return false;
    }

    if (!isNaN(tolerance) && !isNaN(baseline)) {
        const minAllowed = baseline - tolerance;
        const maxAllowed = baseline + tolerance;

        if (value < minAllowed || value > maxAllowed) {
            errorDiv.textContent = `Value ${value} must be between ${minAllowed} and ${maxAllowed}`;
            errorDiv.classList.remove('hidden');
            input.classList.add('border-red-500');
            if (submitBtn) submitBtn.disabled = true;
            return false;
        }
    }

    input.classList.add('border-green-500');
    if (submitBtn) {
        const allValid = checkAllFieldsValid();
        submitBtn.disabled = !allValid;
    }
    return true;
}

// Check if all fields have valid values
function checkAllFieldsValid() {
    const fields = document.querySelectorAll('[id^="input_"]');
    let allValid = true;

    fields.forEach(input => {
        const fieldId = input.id.replace('input_', '');
        const value = parseFloat(input.value);
        const errorDiv = document.getElementById(`error_${fieldId}`);

        if (isNaN(value)) {
            allValid = false;
            if (errorDiv) {
                errorDiv.textContent = 'Please enter a valid number';
                errorDiv.classList.remove('hidden');
                input.classList.add('border-red-500');
            }
            return;
        }

        const field = FIELD_CONFIG.find(f => f.id === fieldId);
        const tolerance = parseFloat(toleranceData[field.tolField]);
        const baseline = parseFloat(baselineData[fieldId]);

        if (!isNaN(tolerance) && !isNaN(baseline)) {
            const minAllowed = baseline - tolerance;
            const maxAllowed = baseline + tolerance;
            if (value < minAllowed || value > maxAllowed) {
                allValid = false;
                if (errorDiv) {
                    errorDiv.textContent = `Value ${value} must be between ${minAllowed} and ${maxAllowed}`;
                    errorDiv.classList.remove('hidden');
                    input.classList.add('border-red-500');
                }
                return;
            }
        }

        if (errorDiv) {
            errorDiv.classList.add('hidden');
        }
        input.classList.remove('border-red-500');
        input.classList.add('border-green-500');
    });

    return allValid;
}

// Submit element data
function submitElementData() {

    let allValid = true;
    let formData = {};

    fields.forEach(input => {
        const fieldId = input.id.replace('input_', '');
        const value = parseFloat(input.value);
        const errorDiv = document.getElementById(`error_${fieldId}`);
        const field = FIELD_CONFIG.find(f => f.id === fieldId);
        const tolerance = parseFloat(toleranceData[field.tolField]);
        const baseline = parseFloat(baselineData[fieldId]);

        if (isNaN(value)) {
            allValid = false;
            if (errorDiv) {
                errorDiv.textContent = 'Please enter a valid number';
                errorDiv.classList.remove('hidden');
                input.classList.add('border-red-500');
            }
            return;
        }

        if (!isNaN(tolerance) && !isNaN(baseline)) {
            const minAllowed = baseline - tolerance;
            const maxAllowed = baseline + tolerance;
            if (value < minAllowed || value > maxAllowed) {
                allValid = false;
                if (errorDiv) {
                    errorDiv.textContent = `Value ${value} must be between ${minAllowed} and ${maxAllowed}`;
                    errorDiv.classList.remove('hidden');
                    input.classList.add('border-red-500');
                }
                return;
            }
        }

        formData[fieldId] = value;
    });

    if (!allValid) {
        alert('Please fix all validation errors before submitting.');
        return;
    }

    // Submit data
    const selectedElement = document.getElementById('kitElementSelect').value;
    const elementDesc = document.getElementById('kitElementSelect').options[document.getElementById('kitElementSelect').selectedIndex].text;

    const payload = {
        element_id: selectedElement,
        element_desc: elementDesc,
        order_code: orderCode,
        demand_code: demandCode,
        pallet_number: palletNumber,
        box_number: boxNumber,
        dimensions: formData
    };

    fetch('/save_element_dimensions/', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': getCookie('csrftoken')
        },
        body: JSON.stringify(payload)
    })
    .then(response => response.json())
    .then(data => {

            location.reload();

        alert('Error saving dimensions. Please try again.');
    });

async function startBackCamera() {
    try {
        console.log('Starting back camera...');
        const constraints = {
            video: {
                facingMode: 'environment',
                width: { ideal: 1280 },
                height: { ideal: 720 }
            }
        };
        
        backStream = await navigator.mediaDevices.getUserMedia(constraints);
        const video = document.getElementById('cameraFeed');
        video.srcObject = backStream;
        
        video.onloadedmetadata = () => {
            video.play()
                .then(() => {
                    console.log('✓ Back camera started');
                    console.log('  Dimensions:', video.videoWidth, 'x', video.videoHeight);
                })
                .catch(e => console.error('Error playing back camera:', e));
        };
    } catch (err) {
        console.error('✗ Back camera error:', err);
        alert('Could not access back camera. Please check permissions.');
    }
}

async function startFrontCamera() {
    try {
        frontStream = await navigator.mediaDevices.getUserMedia({
            video: { facingMode: "user" }

        const video = document.getElementById('frontCameraFeed');
        video.classList.add('hidden');

}

function stopAllCameras() {

        frontStream.getTracks().forEach(track => track.stop());
        frontStream = null;
    }
    frontCameraReady = false;
    frontCameraAttempts = 0;

}

async function captureImage() {
    console.log("📸 STEP 1: Capture BACK image");
    backImageCaptureTime = new Date().toISOString();
    console.log(`⏱️ Back image captured at: ${backImageCaptureTime}`);
    
    const overlay = document.getElementById('captureOverlay');

    const preview = document.getElementById('imagePreview');

async function captureFrontImage() {
    const video = document.getElementById('frontCameraFeed');
    const canvas = document.getElementById('frontCameraCanvas');
    const ctx = canvas.getContext('2d');
    document.getElementById('captureBtn').classList.add('hidden');
    let retries = 0;

    while ((video.videoWidth === 0) && retries < 10) {
        await new Promise(r => setTimeout(r, 200));
        retries++;
    }

    if (!video.videoWidth) {
        console.error("❌ FRONT STILL NOT READY");
        return;
    }

    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    ctx.drawImage(video, 0, 0);

    window.frontImageData = canvas.toDataURL('image/jpeg').split(',')[1];
    frontImageCaptureTime = new Date().toISOString();
    console.log(`⏱️ Front image captured at: ${frontImageCaptureTime}`);

    console.log("✅ FRONT IMAGE CAPTURED");
}

function saveTimingLogToServer(status, additionalData = {}) {

        const end = new Date(uploadCompleteTime);

    }
    
    if (backImageCaptureTime && frontImageCaptureTime) {
        const back = new Date(backImageCaptureTime);
        const front = new Date(frontImageCaptureTime);
        betweenCapturesDuration = Math.abs((front - back) / 1000).toFixed(3);
    }
    
    if (uploadStartTime && uploadCompleteTime) {
        const start = new Date(uploadStartTime);
        const end = new Date(uploadCompleteTime);
        uploadDuration = ((end - start) / 1000).toFixed(3);
    }
    
    const logData = {
        element_desc: elementDesc,
        image_type: additionalData.image_type || (additionalData.is_human ? 'Human Image' : 'Element Image'),
        status: status,
        order_code: orderCode,
        demand_code: demandCode,
        customer_name: customerName,
        customer_po: customerPO,
        pallet_number: palletNumber,
        box_number: boxNumber,
        pl_value: additionalData.pl_value || '1',
        capture_start_time: captureStartTime,
        back_capture_time: backImageCaptureTime,
        front_capture_time: frontImageCaptureTime,
        upload_start_time: uploadStartTime,
        upload_complete_time: uploadCompleteTime,
        total_duration: totalDuration,
        back_capture_duration: backCaptureDuration,
        front_capture_duration: frontCaptureDuration,
        between_captures_duration: betweenCapturesDuration,
        upload_duration: uploadDuration,
        folder_path: additionalData.folder_path || folderPathFromServer || 'N/A',
        ip_address: '',
        error_message: additionalData.error || ''
    };
    
    console.log('📝 Saving timing log to network:', logData);
    
    try {
        fetch('/save_timing_log/', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRFToken': getCookie('csrftoken')
            },
            body: JSON.stringify(logData)
        })
        .then(response => response.json())
        .then(data => {
            if (data.status === 'success') {
                console.log('✅ Timing log saved to: \\\\192.168.4.32\\Corekit\\logs\\');
            } else {
                console.error('❌ Failed to save timing log:', data.message);
            }
        })
        .catch(err => console.error('❌ Error saving timing log:', err));
    } catch (err) {
        console.error('❌ Error saving timing log:', err);
    }
}

function getCookie(name) {
    let cookieValue = null;
    if (document.cookie && document.cookie !== '') {
        const cookies = document.cookie.split(';');
        for (let i = 0; i < cookies.length; i++) {
            const cookie = cookies[i].trim();
            if (cookie.substring(0, name.length + 1) === (name + '=')) {
                cookieValue = decodeURIComponent(cookie.substring(name.length + 1));
                break;
            }
        }
    }
    return cookieValue;
}
